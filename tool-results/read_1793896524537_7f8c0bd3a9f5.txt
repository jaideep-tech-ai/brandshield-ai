   104→function chain(entries: { action: string; actorId: string; details: Record<string, unknown>; ts: Date }[]) {
   105→  let prev = "GENESIS";
   106→  return entries.map((e) => {
   107→    const contentHash = createHash("sha256")
   108→      .update(JSON.stringify({ prevHash: prev, ...e, ts: e.ts.getTime() }))
   109→      .digest("hex");
   110→    const row = { ...e, prevHash: prev, contentHash };
   111→    prev = contentHash;
   112→    return row;
   113→  });
   114→}
